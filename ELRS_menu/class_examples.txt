//The func will be running in separate task with low priority
//Option for canceling the func should be handled here to avoid unexpected problems
void exmp_func(command_obj_t &obj, bool *confirmed, bool *canceled) {
    //starting the func
    //CDM_RESP_EXECUTING - means that the work is in progress
    obj.type = CDM_RESP_EXECUTING;
    ESP_LOGI("tag", "STEP3");
    obj.change_description("Running…", sizeof("Running…"));
    vTaskDelay(pdMS_TO_TICKS(2000));
    //Asking for confirmation from remote
    //the text there can be changed and will be dispaley on top of the pop-up
    obj.change_description("Confirm :)", sizeof("Confirm :)"));
    //this type will evoke pop-up
    obj.type = CDM_RESP_CONFIRM;
    //waiting till remote gives conformation
    while(!(*confirmed)) {
        //if the func is terminated, we just stop it.
        if(*canceled) {
            ESP_LOGW("tag", "cancel.");
            obj.change_description("Canceled.", sizeof("Confirmed"));
            obj.type = CDM_RESP_IDLE;
            return;
        }
        vTaskDelay(20);
    }
    //changing text
    obj.change_description("Confirmed", sizeof("Confirmed"));
    obj.type = CDM_RESP_CONFIRM;
    vTaskDelay(20);
    //pop-up will not dissapear, but user will see "progress" wheel
    obj.type = CDM_RESP_EXECUTING;
    ESP_LOGW("tag", "STEP2");
    vTaskDelay(pdMS_TO_TICKS(2000));
    ESP_LOGE("tag", "STEP1");
    vTaskDelay(pdMS_TO_TICKS(2000));
    ESP_LOGI("tag", "STEP0");
    obj.change_description("Done.", sizeof("Done."));
    str.change_info_text(options_txt[choice_curr], strlen(options_txt[choice_curr]));
    //setting packet type to 0 means that execution of the func ended
    obj.type = CDM_RESP_IDLE;
}
