enum example_command_state_t { READY, RUNNING, WAITING_CONFIRM, DONE };
example_command_state_t command_state = READY;

// Called synchronously from handle_exchange(). It must schedule/change state
// only and return immediately.
void exmp_func(command_obj_t &obj, command_type_e event) {
    switch (event) {
    case CMD_CLICK:
        command_state = RUNNING;
        obj.type = CDM_RESP_EXECUTING;
        obj.change_description("Running…", sizeof("Running…"));
        break;

    case CMD_CONFIRMED:
        command_state = RUNNING;
        obj.type = CDM_RESP_EXECUTING;
        break;

    case CMD_CANCEL:
        command_state = READY;
        obj.type = CDM_RESP_IDLE;
        obj.change_description("Canceled.", sizeof("Canceled."));
        break;

    case CMD_QUERY:
        // Report state when the menu asks. Do not perform long work here.
        if (command_state == WAITING_CONFIRM) {
            obj.type = CDM_RESP_CONFIRM;
            obj.change_description("Confirm :)", sizeof("Confirm :)"));
        } else if (command_state == DONE) {
            obj.type = CDM_RESP_IDLE;
            obj.change_description("Done.", sizeof("Done."));
        }
        break;

    default:
        break;
    }
}

// Run this from the application loop or an application-owned worker.
void command_tick() {
    if (command_state == RUNNING && reset_step_is_complete()) {
        command_state = WAITING_CONFIRM;
    }
}
