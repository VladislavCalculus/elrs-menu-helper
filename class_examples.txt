// C++17 example. Link ELRS_menu::ELRS_menu and include
// <elrs_menu/elrs_menu.hpp> in the application.

struct CrsfTransport {
    void write(const uint8_t *data, size_t size);
};

void elrs_menu_write(const uint8_t *data, size_t size, void *context) {
    static_cast<CrsfTransport *>(context)->write(data, size);
}

const char status_text[] = "Ready";
info_obj_t status_value{status_text, sizeof(status_text)};

uint8_t selected_mode = 0;
const char mode_options[] = "Normal;Sport;";
text_select_obj_t mode_value{&selected_mode, mode_options, sizeof(mode_options)};

int16_t limit = 100;
int_obj_t limit_value{&limit, sizeof(limit), true, -500, 500, 100};

enum class reset_state_t { idle, running, waiting_for_confirmation, complete };
reset_state_t reset_state = reset_state_t::idle;

// Called from handle_exchange(). Do not block here: schedule or advance work
// in the application's event loop/task instead.
void reset_handler(command_obj_t &command, command_type_e event) {
    switch (event) {
    case CMD_CLICK:
        reset_state = reset_state_t::running;
        command.type = CDM_RESP_EXECUTING;
        command.change_description("Resetting", sizeof("Resetting"));
        break;
    case CMD_CONFIRMED:
        reset_state = reset_state_t::running;
        command.type = CDM_RESP_EXECUTING;
        break;
    case CMD_CANCEL:
        reset_state = reset_state_t::idle;
        command.type = CDM_RESP_IDLE;
        command.change_description("Canceled", sizeof("Canceled"));
        break;
    case CMD_QUERY:
        if (reset_state == reset_state_t::waiting_for_confirmation) {
            command.type = CDM_RESP_CONFIRM;
            command.change_description("Confirm reset?", sizeof("Confirm reset?"));
        } else if (reset_state == reset_state_t::complete) {
            command.type = CDM_RESP_IDLE;
            command.change_description("Done", sizeof("Done"));
        }
        break;
    default:
        break;
    }
}

command_obj_t reset_value{"Reset device", sizeof("Reset device"), 200, reset_handler};

// Folder IDs are parameter-array indices plus one. This folder is entry 1.
param_entry_t parameters[] = {
    {"Settings", nullptr, PARENT_ENTRY_ROUTE_GLOBAL, CRSF_PARAM_TYPE_FOLDER},
    {"Mode", &mode_value, 1, CRSF_PARAM_TYPE_TEXT_SELECTION},
    {"Limit", &limit_value, 1, CRSF_PARAM_TYPE_INT16},
    {"Status", &status_value, PARENT_ENTRY_ROUTE_GLOBAL, CRSF_PARAM_TYPE_INFO},
    {"Reset", &reset_value, PARENT_ENTRY_ROUTE_GLOBAL, CRSF_PARAM_TYPE_COMMAND},
};

CrsfTransport transport;
elrs_menu_config_t config{};
config.device_info = {"Virtual Insanity", 0x434F4E56, 0x01000000, 0x00010203, 2};
config.parameters = parameters;
config.parameter_count = sizeof(parameters) / sizeof(parameters[0]);
config.write = elrs_menu_write;
config.write_context = &transport;

elrs_menu_t menu{config};

// The CRSF parser calls this only for an assembled, CRC-valid frame.
void on_crsf_frame(const uint8_t *frame, size_t frame_size) {
    handle_exchange(menu, frame, frame_size);
}
